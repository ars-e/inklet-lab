// Flippable POI Cards with lazy preview (image by default) document.querySelectorAll('.poi-card').forEach(card => { const link = card.querySelector('.preview-link'); const href = card.dataset.href; const previewSrc = card.dataset.preview; // Flip card (ignore clicks on the preview) card.addEventListener('click', (e) => { if (e.target.closest('.preview-link')) return; card.classList.toggle('is-flipped'); // Lazy load preview once (image) or swap to iframe if you want live view if (card.classList.contains('is-flipped') && link && !link.dataset.loaded) { const img = new Image(); img.alt = ''; img.loading = 'lazy'; img.src = previewSrc; img.onerror = () => { link.style.background = 'linear-gradient(180deg,#2f2f2f,#1e1e1e)'; }; link.appendChild(img); link.dataset.loaded = '1'; } }); // Click preview → go to page (don’t trigger flip) if (link && href) { link.href = href; link.addEventListener('click', (e) => e.stopPropagation()); } }); /* If you prefer live page in the back: Replace the Image() block with: const frame = document.createElement('iframe'); frame.src = href; frame.title = 'Preview'; frame.loading = 'lazy'; frame.sandbox = 'allow-same-origin allow-scripts'; link.appendChild(frame); link.dataset.loaded = '1'; */
